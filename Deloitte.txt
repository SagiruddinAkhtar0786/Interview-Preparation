The @Primary and @Qualifier annotations in Spring are both used for resolving bean injection conflicts, but they work in different ways. Here’s a comparison between the two:

@Primary   :annotation is used when there are multiple beans of the same type and you want to inject the bean byDefault then we use 

@Configuration
class BeanConfiguration {

    @Bean
    @Primary
    public DataSource primary() {
        return new DataSource();
    }

    @Bean
    public DataSource secondary() {
        return new DataSource();
    }
}

///////////
@Qualifier : when we have the multiple beans of  same type and we want to use specific bean for a particular place
then we use @Qualifier annotation

@Configuration
class ConfigurationClass {

    @Bean
    public DataSource primary() {
        return new DataSource();
    }

    @Bean
    public DataSource secondary() {
        return new DataSource();
    }
}

@Component
class ComponentClass {

    private final DataSource ds;

    public ComponentClass(@Qualifier("secondary") DataSource dataS) {
        this.ds = dataS;
    }

    public DataSource getDataSource() {
        return ds;
    }


}

package DATASTRUCTURE_Implemetation;

class LinkedList {

    // Node class
    private static class Node {
        int data;
        Node next;

        Node(int data) {
            this.data = data;
            this.next = null;
        }
    }

    private Node head; // Head of the list

    // Add a node at the end
    public void add(int data) {
        Node newNode = new Node(data);
        if (head == null) {
            head = newNode;
            return;
        }
        Node current = head;
        while (current.next != null) {
            current = current.next;
        }
        current.next = newNode;
    }

    // Delete a node with a specific value
    public boolean delete(int data) {
        if (head == null) {
            return false; // List is empty
        }
        if (head.data == data) { // If the head node is to be deleted
            head = head.next;
            return true;
        }
        Node current = head;
        while (current.next != null && current.next.data != data) {
            current = current.next;
        }
        if (current.next == null) {
            return false; // Data not found
        }
        current.next = current.next.next;
        return true;
    }

    // Display the linked list
    public void display() {
        if (head == null) {
            System.out.println("List is empty");
            return;
        }
        Node current = head;
        while (current != null) {
            System.out.print(current.data + " -> ");
            current = current.next;
        }
        System.out.println("null");
    }

    // Reverse the linked list
    public void reverse() {
        Node previous = null;
        Node current = head;
        Node next = null;

        while (current != null) {
            next = current.next; // Store next node
            current.next = previous; // Reverse the current node's pointer
            previous = current; // Move previous to current
            current = next; // Move current to next
        }
        head = previous; // Update head to the last node
    }

    // Main method to test the linked list
    public static void main(String[] args) {
        LinkedList list = new LinkedList();

        // Adding elements
        list.add(10);
        list.add(20);
        list.add(30);
        list.add(40);
        System.out.println("Original List:");
        list.display();

        // Deleting an element
        list.delete(20);
        System.out.println("After Deletion of 20:");
        list.display();

        // Reversing the list
        list.reverse();
        System.out.println("After Reversing:");
        list.display();
    }
}
